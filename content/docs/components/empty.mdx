## Usage

```tsx
import {
  Empty,
  EmptyContent,
  EmptyDescription,
  EmptyHeader,
  EmptyMedia,
  EmptyTitle,
} from "@/components/ui/empty"
```

```tsx
<Empty>
  <EmptyHeader>
    <EmptyMedia>
      <CalendarXIcon />
    </EmptyMedia>
    <EmptyTitle>No bookings yet</EmptyTitle>
    <EmptyDescription>Share your booking page.</EmptyDescription>
  </EmptyHeader>
  <EmptyContent>
    <Button>Share booking page</Button>
  </EmptyContent>
</Empty>
```

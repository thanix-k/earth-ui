## Usage

```tsx
import {
  Sheet,
  SheetContent,
  SheetDescription,
  SheetHeader,
  SheetTitle,
  SheetTrigger,
} from "@/components/ui/sheet"
```

```tsx
<Sheet>
  <SheetTrigger>Open</SheetTrigger>
  <SheetContent>
    <SheetHeader>
      <SheetTitle>Edit staff</SheetTitle>
      <SheetDescription>Update how Jordan appears on your booking page.</SheetDescription>
    </SheetHeader>
  </SheetContent>
</Sheet>
```

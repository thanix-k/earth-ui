## Usage

```tsx
import {
  Field,
  FieldDescription,
  FieldError,
  FieldLabel,
} from "@/components/ui/field"
```

```tsx
<Field>
  <FieldLabel htmlFor="email">Email</FieldLabel>
  <Input id="email" />
  <FieldDescription>We send booking confirmations here.</FieldDescription>
</Field>
```

Group fields with `FieldGroup`, and related fields under a legend with `FieldSet` and `FieldLegend`.

## Usage

```tsx
import {
  Drawer,
  DrawerContent,
  DrawerDescription,
  DrawerHeader,
  DrawerTitle,
  DrawerTrigger,
} from "@/components/ui/drawer"
```

```tsx
<Drawer>
  <DrawerTrigger>Open</DrawerTrigger>
  <DrawerContent>
    <DrawerHeader>
      <DrawerTitle>Haircut & style</DrawerTitle>
      <DrawerDescription>Pick a time for your appointment.</DrawerDescription>
    </DrawerHeader>
  </DrawerContent>
</Drawer>
```

Set `showSwipeHandle` on `Drawer` to show the grab handle.

## Examples

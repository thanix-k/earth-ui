## Usage

```tsx
import { Checkbox } from "@/components/ui/checkbox"
import { Label } from "@/components/ui/label"
```

```tsx
<Label>
  <Checkbox />
  Send reminders
</Label>
```

Pass `indeterminate` for a parent checkbox whose children are partly selected.

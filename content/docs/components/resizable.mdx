## Usage

```tsx
import {
  ResizableHandle,
  ResizablePanel,
  ResizablePanelGroup,
} from "@/components/ui/resizable"
```

```tsx
<ResizablePanelGroup orientation="horizontal">
  <ResizablePanel>Services</ResizablePanel>
  <ResizableHandle />
  <ResizablePanel>Service details</ResizablePanel>
</ResizablePanelGroup>
```

## Usage

```tsx
import {
  ContextMenu,
  ContextMenuContent,
  ContextMenuItem,
  ContextMenuTrigger,
} from "@/components/ui/context-menu"
```

```tsx
<ContextMenu>
  <ContextMenuTrigger>Right-click here</ContextMenuTrigger>
  <ContextMenuContent>
    <ContextMenuItem>New appointment</ContextMenuItem>
    <ContextMenuItem>Block time</ContextMenuItem>
  </ContextMenuContent>
</ContextMenu>
```

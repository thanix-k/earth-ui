## Usage

```tsx
import { Avatar, AvatarFallback, AvatarImage } from "@/components/ui/avatar"
```

```tsx
<Avatar>
  <AvatarImage src="https://github.com/shadcn.png" alt="Maya Patel" />
  <AvatarFallback>MP</AvatarFallback>
</Avatar>
```

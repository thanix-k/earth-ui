## Usage

```tsx
import { NativeSelect, NativeSelectOption } from "@/components/ui/native-select"
```

```tsx
<NativeSelect>
  <NativeSelectOption value="30">30 minutes</NativeSelectOption>
  <NativeSelectOption value="60">1 hour</NativeSelectOption>
</NativeSelect>
```

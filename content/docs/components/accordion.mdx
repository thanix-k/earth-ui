## Usage

```tsx
import {
  Accordion,
  AccordionContent,
  AccordionItem,
  AccordionTrigger,
} from "@/components/ui/accordion"
```

```tsx
<Accordion>
  <AccordionItem value="item-1">
    <AccordionTrigger>Can customers reschedule?</AccordionTrigger>
    <AccordionContent>Yes, up to 24 hours before.</AccordionContent>
  </AccordionItem>
</Accordion>
```

## Usage

```tsx
import {
  Menubar,
  MenubarContent,
  MenubarItem,
  MenubarMenu,
  MenubarTrigger,
} from "@/components/ui/menubar"
```

```tsx
<Menubar>
  <MenubarMenu>
    <MenubarTrigger>Booking</MenubarTrigger>
    <MenubarContent>
      <MenubarItem>New appointment</MenubarItem>
      <MenubarItem>New class</MenubarItem>
    </MenubarContent>
  </MenubarMenu>
</Menubar>
```

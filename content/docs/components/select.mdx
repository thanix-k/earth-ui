## Usage

```tsx
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from "@/components/ui/select"
```

```tsx
const items = [
  { label: "Alex Morgan", value: "alex" },
  { label: "Priya Shah", value: "priya" },
]

<Select items={items}>
  <SelectTrigger className="w-56">
    <SelectValue placeholder="Select staff" />
  </SelectTrigger>
  <SelectContent>
    {items.map((item) => (
      <SelectItem key={item.value} value={item.value}>
        {item.label}
      </SelectItem>
    ))}
  </SelectContent>
</Select>
```

The menu opens below the trigger, as in Earth. Pass `alignItemWithTrigger` to `SelectContent` to overlay the selected item on the trigger instead.

## Usage

```tsx
import { Label } from "@/components/ui/label"
import { Switch } from "@/components/ui/switch"
```

```tsx
<Label>
  <Switch />
  Accept online bookings
</Label>
```

The switch is 36×20px on touch screens and 28×16px from the `lg` breakpoint.

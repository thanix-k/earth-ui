## Usage

```tsx
import {
  Command,
  CommandEmpty,
  CommandGroup,
  CommandInput,
  CommandItem,
  CommandList,
} from "@/components/ui/command"
```

```tsx
<Command>
  <CommandInput placeholder="Search…" />
  <CommandList>
    <CommandEmpty>No results found.</CommandEmpty>
    <CommandGroup heading="Suggestions">
      <CommandItem>New appointment</CommandItem>
      <CommandItem>Working hours</CommandItem>
    </CommandGroup>
  </CommandList>
</Command>
```

Built on [cmdk](https://github.com/pacocoursey/cmdk).

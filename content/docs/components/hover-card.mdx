## Usage

```tsx
import { HoverCard, HoverCardContent, HoverCardTrigger } from "@/components/ui/hover-card"
```

```tsx
<HoverCard>
  <HoverCardTrigger href="/staff/jordan">Jordan Lee</HoverCardTrigger>
  <HoverCardContent>Senior stylist.</HoverCardContent>
</HoverCard>
```

## Usage

```tsx
import { Label } from "@/components/ui/label"
import { RadioGroup, RadioGroupItem } from "@/components/ui/radio-group"
```

```tsx
<RadioGroup defaultValue="24h">
  <Label>
    <RadioGroupItem value="1h" />
    1 hour before
  </Label>
  <Label>
    <RadioGroupItem value="24h" />
    24 hours before
  </Label>
</RadioGroup>
```

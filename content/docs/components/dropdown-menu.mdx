## Usage

```tsx
import {
  DropdownMenu,
  DropdownMenuContent,
  DropdownMenuItem,
  DropdownMenuTrigger,
} from "@/components/ui/dropdown-menu"
```

```tsx
<DropdownMenu>
  <DropdownMenuTrigger>Open</DropdownMenuTrigger>
  <DropdownMenuContent>
    <DropdownMenuItem>Edit</DropdownMenuItem>
    <DropdownMenuItem>Reschedule</DropdownMenuItem>
  </DropdownMenuContent>
</DropdownMenu>
```

Items are 14px with more padding on touch screens, and 12px from the `lg` breakpoint. Use `variant="destructive"` for actions that delete or cancel.

## Usage

```tsx
import {
  Item,
  ItemActions,
  ItemContent,
  ItemDescription,
  ItemMedia,
  ItemTitle,
} from "@/components/ui/item"
```

```tsx
<Item variant="outline">
  <ItemContent>
    <ItemTitle>Online booking</ItemTitle>
    <ItemDescription>Customers can book any open slot.</ItemDescription>
  </ItemContent>
  <ItemActions>
    <Button variant="outline" size="sm">Settings</Button>
  </ItemActions>
</Item>
```
